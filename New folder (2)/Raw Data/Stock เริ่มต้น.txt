ไม่ใช่ที่นี่ (แก้ที่นี่แล้วไม่มีผลกับเกมจริง) ส่วน "สรุปเงินทุน" ด้านล่างยังเป็น
คำอธิบายที่มาของตัวเลข ไม่ใช่ข้อมูลดิบ เก็บไว้อ่านได้ตามปกติ

const stockMax = {
  flour: 10,        // kg
  yeast: 1,         // kg
  sugar: 3.5,       // kg
  milk: 5,          // L
  butter: 1.2,      // kg
  egg: 30,          // ฟอง
  redbean: 2.0,     // kg (Patch 4.3.4: 0.5 -> 2.0 — buffer เดิมคุ้มแค่ ~1.5-2 วัน
                     //     ในขณะที่วัตถุดิบอื่นมี buffer ~7 วัน ทำให้เป็นคอขวดซ้ำแทบทุกวัน)
  cocoa: 0          // g
};

// unitScale (Patch 5.0.2): ตัวคูณแปลงหน่วยเก็บ -> หน่วยคำนวณสูตร ต่อวัตถุดิบ
// kg/L -> g/mL ใช้ 1000, ส่วนไข่ (นับเป็นฟอง) กับผงโกโก้ (เก็บเป็นกรัมอยู่แล้ว) ใช้ 1
// วัตถุดิบใหม่ทุกตัวต้องประกาศที่นี่ด้วย ไม่งั้น engine จะ throw ทันทีตอนโหลด
const unitScale = {
  flour: 1000, yeast: 1000, sugar: 1000,
  milk: 1000, butter: 1000, redbean: 1000,
  egg: 1, cocoa: 1
};

// Patch 4.3.3: reorderThreshold ต่อวัตถุดิบแทนที่ 10% คงที่ทั้งหมด
// Patch 4.4.1: egg 0.10 -> 0.20 (กันไข่ลงไปเหลือใกล้ 0 ก่อนโดนเติมเต็มแผงทีเดียว)
const reorderThreshold = {
  flour: 0.10, yeast: 0.10, sugar: 0.10,
  milk: 0.15, butter: 0.15, redbean: 0.30,
  egg: 0.20, cocoa: 0.10
};

const stock = { ...stockMax }; // วันเปิดร้าน stock = stockMax
